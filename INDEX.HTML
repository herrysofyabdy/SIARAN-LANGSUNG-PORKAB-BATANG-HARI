Masuk dengan kode pengelola, * lalu buat kode untuk operator meja di panel Kontrol. * * Link layar videotron (tanpa tombol): URL/exec?layar=1 * Link tonton live untuk penonton (tanpa login): URL/exec?tonton=1 * Cek API: buka URL/exec?ping -> harus tampil {"ok":true,...} * * index.html juga bisa dipasang di GitHub Pages: isi konstanta API di index.html * dengan URL /exec ini. Halaman memanggil doPost di bawah. * * Notasi setiap partai otomatis disimpan ke Google Sheet * "Arsip Notasi Papan Catur Live" di Google Drive Anda * (dibuat otomatis saat partai pertama disimpan). * Saat deploy pertama, Google akan meminta izin akses Spreadsheet: izinkan. */ const KODE_PENGELOLA_AWAL = 'GANTI-KODE-INI'; const MAX_PAPAN = 12; /* ------------------------------------------------------------------ */ /** * Jalankan SEKALI dari editor (pilih "setup" lalu klik Jalankan). * Meminta izin Google, membuat spreadsheet arsip notasi, dan menyiapkan data awal. * Hasilnya bisa dilihat di Log eksekusi. */ function setup() { if (!props().getProperty('META')) props().setProperty('META', JSON.stringify(defMeta())); archSheet(true); cache().remove('STATE'); cache().remove('AUTH'); Logger.log('Siap. Arsip notasi: ' + archiveUrl()); Logger.log('Kode pengelola aktif: ' + readAuth().admin); Logger.log('Langkah berikutnya: Terapkan > Deployment baru > Aplikasi web.'); } /** * API untuk index.html yang dipasang di luar Apps Script (GitHub Pages, hosting lain, atau dibuka dari laptop). * Halaman mengirim POST berisi {"fn":"getState","args":[...]} ke URL /exec. */ function doPost(e) { let out; try { const req = JSON.parse((e && e.postData && e.postData.contents) || '{}'); const fns = { login: login, getState: getState, act: act, getArchive: getArchive }; const f = fns[req.fn]; if (!f) throw new Error('Fungsi tidak dikenal: ' + req.fn); out = f.apply(null, req.args || []); } catch (err) { out = { __err: err.message }; } return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { const p = (e && e.parameter) || {}; if ('ping' in p) { return ContentService.createTextOutput(JSON.stringify({ ok: true, aplikasi: 'Papan Catur Live' })) .setMimeType(ContentService.MimeType.JSON); } const t = HtmlService.createTemplateFromFile('index'); t.boot = JSON.stringify({ mode: ('layar' in p) ? 'layar' : ('tonton' in p) ? 'tonton' : '', url: ScriptApp.getService().getUrl() }); return t.evaluate() .setTitle('Papan Catur Live') .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover') .addMetaTag('mobile-web-app-capable', 'yes') .addMetaTag('apple-mobile-web-app-capable', 'yes') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } /* ---------------- penyimpanan ---------------- */ const props = () => PropertiesService.getScriptProperties(); const cache = () => CacheService.getScriptCache(); function defMeta() { return { ev: 'Kejuaraan Catur', rd: 'Babak 1', n: 2, tc: { min: 90, inc: 30 }, showClock: true, pub: true, qr: false, ver: 1 }; } function newClock(tc) { return { w: tc.min * 60000, b: tc.min * 60000, run: false, since: 0 }; } function newGid() { return Utilities.getUuid().replace(/-/g, '').slice(0, 10); } function defBoard(k, tc) { return { label: 'Meja ' + (k + 1), w: '', wi: '', b: '', bi: '', mv: [], result: null, showRes: true, flip: false, clock: newClock(tc), gid: newGid() }; } function readMeta() { const s = props().getProperty('META'); return s ? JSON.parse(s) : defMeta(); } function readBoard(k, meta) { const s = props().getProperty('B' + k); return s ? JSON.parse(s) : defBoard(k, meta.tc); } function saveBoard(k, b) { props().setProperty('B' + k, JSON.stringify(b)); } function buildState() { const m = readMeta(), boards = []; for (let k = 0; k < m.n; k++) boards.push(readBoard(k, m)); return { ev: m.ev, rd: m.rd, n: m.n, tc: m.tc, showClock: m.showClock, pub: m.pub, qr: !!m.qr, ver: m.ver, boards: boards }; } /* ---------------- arsip notasi (Google Sheet) ---------------- */ const ARCH_HEAD = ['ID', 'Disimpan', 'Acara', 'Babak', 'Meja', 'Putih', 'Hitam', 'Hasil', 'Keterangan', 'Langkah', 'Notasi', 'PGN']; function archSheet(create) { const id = props().getProperty('ARCHIVE_ID'); let ss = null; if (id) { try { ss = SpreadsheetApp.openById(id); } catch (e) { ss = null; } } if (!ss) { if (!create) return null; ss = SpreadsheetApp.create('Arsip Notasi Papan Catur Live'); const sh0 = ss.getSheets()[0]; sh0.getRange(1, 1, 1, ARCH_HEAD.length).setValues([ARCH_HEAD]); sh0.setFrozenRows(1); props().setProperty('ARCHIVE_ID', ss.getId()); } return ss.getSheets()[0]; } function archiveUrl() { const id = props().getProperty('ARCHIVE_ID'); return id ? 'https://docs.google.com/spreadsheets/d/' + id + '/edit' : ''; } /** Simpan / perbarui satu partai (satu baris per ID partai). */ function archUpsert(meta, b, save) { if (!save || !save.pgn || !b.mv.length) return false; const sh = archSheet(true); const row = [b.gid, new Date(), meta.ev, meta.rd, b.label, b.w || 'Putih', b.b || 'Hitam', (b.result && b.result.score) || '*', (b.result && b.result.why) || 'Belum selesai', Math.ceil(b.mv.length / 2), String(save.san || '').slice(0, 45000), String(save.pgn).slice(0, 45000)]; const n = sh.getLastRow(); let r = 0; if (n > 1) { const ids = sh.getRange(2, 1, n - 1, 1).getValues(); for (let i = ids.length - 1; i >= 0; i--) if (ids[i][0] === b.gid) { r = i + 2; break; } } if (r) sh.getRange(r, 1, 1, row.length).setValues([row]); else sh.appendRow(row); cache().remove('ARCH'); return true; } function getArchive(code) { const st = getStateObj(); if (!st.pub && !roleOf(code)) return { err: 'auth' }; const c = cache().get('ARCH'); if (c) return { list: JSON.parse(c) }; const sh = archSheet(false); let list = []; if (sh && sh.getLastRow() > 1) { const v = sh.getRange(2, 1, sh.getLastRow() - 1, ARCH_HEAD.length).getValues(); list = v.map(r => ({ id: r[0], at: (r[1] instanceof Date) ? r[1].getTime() : r[1], ev: r[2], rd: r[3], lb: r[4], w: r[5], b: r[6], res: r[7], why: r[8], n: r[9], pgn: r[11] })) .reverse().slice(0, 300); } try { cache().put('ARCH', JSON.stringify(list), 21600); } catch (e) {} return { list: list }; } function cacheState(st) { try { cache().put('STATE', JSON.stringify(st), 21600); } catch (e) {} } function getStateObj() { const c = cache().get('STATE'); if (c) return JSON.parse(c); const st = buildState(); cacheState(st); return st; } /* ---------------- akses ---------------- */ function readAuth() { const c = cache().get('AUTH'); if (c) return JSON.parse(c); const a = { admin: (props().getProperty('ADMIN_CODE') || KODE_PENGELOLA_AWAL).toUpperCase(), codes: JSON.parse(props().getProperty('CODES') || '[]') }; cache().put('AUTH', JSON.stringify(a), 21600); return a; } function roleOf(code) { code = String(code || '').trim().toUpperCase(); if (!code) return null; const a = readAuth(); if (code === a.admin) return { role: 'admin', name: 'Pengelola', boards: 'all' }; const c = a.codes.filter(x => x.code === code)[0]; return c ? { role: 'op', name: c.name, boards: c.boards } : null; } function cleanCodes(list, adminCode) { const seen = {}, out = []; (list || []).forEach(x => { const code = String(x.code || '').trim().toUpperCase(); if (code.length < 4 || code === adminCode || seen[code]) return; seen[code] = 1; let boards = 'all'; if (Array.isArray(x.boards)) boards = x.boards.map(Number).filter(n => n >= 0 && n < MAX_PAPAN); out.push({ name: String(x.name || 'Operator').slice(0, 40), code: code, boards: boards }); }); return out; } /* ---------------- dipanggil dari halaman ---------------- */ function login(code) { const who = roleOf(code); if (!who) return { err: 'Kode akses tidak dikenal.' }; if (who.role === 'admin') { who.codes = readAuth().codes; who.archiveUrl = archiveUrl(); } who.now = Date.now(); who.state = getStateObj(); return who; } function getState(code, ver) { const st = getStateObj(); if (!st.pub && !roleOf(code)) return { err: 'auth', now: Date.now() }; if (ver === st.ver) return { now: Date.now(), ver: st.ver }; return { now: Date.now(), ver: st.ver, state: st }; } function act(code, op) { const who = roleOf(code); if (!who) return { err: 'Kode akses tidak berlaku lagi. Silakan masuk ulang.', relogin: true, now: Date.now() }; const lock = LockService.getScriptLock(); try { lock.waitLock(15000); } catch (e) { return { err: 'Server sedang sibuk, coba lagi.', now: Date.now() }; } try { const meta = readMeta(), now = Date.now(), extra = {}; const adminOnly = () => { if (who.role !== 'admin') throw new Error('Hanya pengelola yang bisa mengubah ini.'); }; let k = null, b = null; if (op.k !== undefined && op.k !== null) { k = Number(op.k); if (!(k >= 0 && k < meta.n)) throw new Error('Papan tidak ada.'); if (who.boards !== 'all' && who.boards.indexOf(k) < 0) throw new Error('Kode Anda tidak punya akses ke papan ini.'); b = readBoard(k, meta); if (!b.gid) b.gid = newGid(); } const archive = (bb, save) => { try { if (archUpsert(meta, bb, save)) { extra.saved = true; extra.archiveUrl = archiveUrl(); } } catch (e) { extra.archErr = 'Notasi gagal disimpan ke Spreadsheet: ' + e.message; } }; const turn = () => (b.mv.length % 2 === 0 ? 'w' : 'b'); const stop = () => { const c = b.clock; if (c.run) { const t = turn(); c[t] = Math.max(0, c[t] - (now - c.since)); c.run = false; } }; const start = () => { if (!b.result) { b.clock.run = true; b.clock.since = now; } }; switch (op.t) { case 'move': { if (b.result) throw new Error('Partai di papan ini sudah selesai.'); if (b.mv.length !== op.ply) throw new Error('Langkah ditolak: papan ini baru saja diubah dari perangkat lain.'); if (!/^[a-h][1-8][a-h][1-8][qrbn]?$/.test(op.m)) throw new Error('Format langkah salah.'); const c = b.clock, t = turn(); if (c.run) { c[t] = Math.max(0, c[t] - (now - c.since)) + (meta.tc.inc || 0) * 1000; c.since = now; } b.mv.push(op.m); if (op.result) { b.result = op.result; b.showRes = true; stop(); archive(b, op.save); } break; } case 'undo': if (b.result) b.result = null; else if (b.mv.length) { const was = b.clock.run; stop(); b.mv.pop(); if (was) start(); } break; case 'board': ['label', 'w', 'wi', 'b', 'bi'].forEach(f => { if (op.v[f] !== undefined) b[f] = String(op.v[f]).slice(0, 80); }); if (op.v.flip !== undefined) b.flip = !!op.v.flip; break; case 'clock': if (op.a === 'start') start(); else if (op.a === 'stop') stop(); else if (op.a === 'set') { const was = b.clock.run; stop(); if (op.w >= 0) b.clock.w = op.w; if (op.b >= 0) b.clock.b = op.b; if (was) start(); } break; case 'result': stop(); b.result = op.v || null; if (op.v) { b.showRes = true; archive(b, op.save); } break; case 'showRes': b.showRes = !!op.v; break; case 'save': if (!b.mv.length) throw new Error('Belum ada langkah untuk disimpan.'); archive(b, op.save); break; case 'new': if (b.mv.length) archive(b, op.save); b.mv = []; b.result = null; b.showRes = true; b.clock = newClock(meta.tc); b.gid = newGid(); break; case 'load': stop(); b.gid = newGid(); b.mv = (op.mv || []).filter(m => /^[a-h][1-8][a-h][1-8][qrbn]?$/.test(m)); b.result = op.result || null; b.showRes = true; if (op.w) b.w = String(op.w).slice(0, 80); if (op.b) b.b = String(op.b).slice(0, 80); break; case 'meta': adminOnly(); if (op.v.ev !== undefined) meta.ev = String(op.v.ev).slice(0, 120); if (op.v.rd !== undefined) meta.rd = String(op.v.rd).slice(0, 60); break; case 'n': adminOnly(); meta.n = Math.max(1, Math.min(MAX_PAPAN, Number(op.v) || 1)); break; case 'tc': adminOnly(); meta.tc = { min: Number(op.min) > 0 ? Number(op.min) : 90, inc: Number(op.inc) >= 0 ? Number(op.inc) : 0 }; for (let i = 0; i < meta.n; i++) { const bb = readBoard(i, meta); bb.clock = newClock(meta.tc); saveBoard(i, bb); } break; case 'showClock': adminOnly(); meta.showClock = !!op.v; break; case 'pub': adminOnly(); meta.pub = !!op.v; break; case 'qr': adminOnly(); meta.qr = !!op.v; break; case 'codes': { adminOnly(); const codes = cleanCodes(op.v, readAuth().admin); props().setProperty('CODES', JSON.stringify(codes)); cache().remove('AUTH'); extra.codes = codes; break; } case 'adminCode': { adminOnly(); const v = String(op.v || '').trim().toUpperCase(); if (v.length < 6) throw new Error('Kode pengelola minimal 6 karakter.'); if (readAuth().codes.some(c => c.code === v)) throw new Error('Kode itu sudah dipakai operator.'); props().setProperty('ADMIN_CODE', v); cache().remove('AUTH'); extra.newCode = v; break; } case 'resetAll': adminOnly(); for (let i = 0; i < meta.n; i++) { const bb = readBoard(i, meta); if (!bb.gid) bb.gid = newGid(); if (bb.mv.length) archive(bb, (op.saves || {})[i]); bb.mv = []; bb.result = null; bb.showRes = true; bb.clock = newClock(meta.tc); bb.gid = newGid(); if (op.names) { bb.w = ''; bb.wi = ''; bb.b = ''; bb.bi = ''; } saveBoard(i, bb); } break; default: throw new Error('Perintah tidak dikenal.'); } if (b) saveBoard(k, b); meta.ver = (meta.ver || 1) + 1; props().setProperty('META', JSON.stringify(meta)); const st = buildState(); cacheState(st); return Object.assign({ now: Date.now(), ver: st.ver, state: st }, extra); } catch (e) { return { err: e.message, now: Date.now(), state: getStateObj() }; } finally { lock.releaseLock(); } }
